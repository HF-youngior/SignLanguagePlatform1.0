## 使用方法

### 1. 翻译功能使用示例

在 `Translate.vue` 中添加：

```javascript
import apiService from '@/services/api'

// 保存翻译记录
async saveTranslationToHistory() {
  try {
    await apiService.saveTranslation({
      type: 'image', // 或 'video', 'camera'
      detectedSigns: this.detectionResults,
      originalFile: {
        fileName: this.currentFile.name,
        fileType: this.currentFile.type,
        fileSize: this.currentFile.size
      },
      metadata: {
        width: this.imageWidth,
        height: this.imageHeight
      },
      processingTime: this.inferenceTime,
      confidence: this.confidenceThreshold
    })
    
    ElMessage.success('翻译记录已保存！')
  } catch (error) {
    console.error('保存失败:', error)
    ElMessage.error('保存失败，请重试')
  }
}

// 获取翻译历史
async loadTranslationHistory() {
  try {
    const response = await apiService.getTranslationHistory({
      page: 1,
      limit: 20,
      type: 'image' // 可选：'image', 'video', 'camera'
    })
    
    this.historyRecords = response.data.data
  } catch (error) {
    console.error('加载历史失败:', error)
  }
}
```

### 2. 社区功能使用示例

在 `Community.vue` 中，将硬编码的数据替换为API调用：

```javascript
import apiService from '@/services/api'

// 获取帖子列表
async loadPosts() {
  try {
    const response = await apiService.getPosts({
      page: 1,
      limit: 20,
      category: '学习心得' // 可选
    })
    
    this.posts = response.data.data
  } catch (error) {
    console.error('加载帖子失败:', error)
  }
}

// 发布新帖子
async publishPost() {
  try {
    const hashtags = this.extractHashtags(this.newPost)
    
    await apiService.createPost({
      title: '我的学习心得',
      content: this.newPost,
      category: '学习心得',
      hashtags,
      images: this.uploadedImages.map(img => img.url),
      videos: this.uploadedVideos.map(vid => vid.url)
    })
    
    ElMessage.success('发布成功！')
    this.newPost = ''
    this.uploadedImages = []
    this.uploadedVideos = []
    await this.loadPosts() // 重新加载帖子列表
  } catch (error) {
    console.error('发布失败:', error)
    ElMessage.error('发布失败，请重试')
  }
}

// 点赞帖子
async likePost(postId) {
  try {
    await apiService.likePost(postId)
    // 更新本地状态或重新加载数据
  } catch (error) {
    console.error('点赞失败:', error)
  }
}
```

## 扩展功能

### 翻译历史页面

可以创建一个新页面显示所有翻译历史：

```vue
<template>
  <div class="translation-history">
    <h1>翻译历史</h1>
    <el-table :data="historyRecords">
      <el-table-column prop="type" label="类型" />
      <el-table-column prop="createdAt" label="时间" />
      <el-table-column prop="detectedSigns" label="检测到的内容" />
      <el-table-column>
        <template #default="scope">
          <el-button @click="viewDetail(scope.row)">查看详情</el-button>
          <el-button @click="deleteRecord(scope.row._id)">删除</el-button>
        </template>
      </el-table-column>
    </el-table>
  </div>
</template>

<script>
import apiService from '@/services/api'

export default {
  data() {
    return {
      historyRecords: []
    }
  },
  async mounted() {
    await this.loadHistory()
  },
  methods: {
    async loadHistory() {
      const response = await apiService.getTranslationHistory()
      this.historyRecords = response.data.data
    },
    async deleteRecord(id) {
      await apiService.deleteTranslation(id)
      await this.loadHistory()
    }
  }
}
</script>
```
